function runSimulation() { const n = parseInt(document.getElementById('apexInput').value); // سقف الحماية للهاتف والمتصفح if (isNaN(n) || n < 2) { alert(currentLang === 'ar' ? 'المرجو إدخال عدد صحيح أكبر من أو يساوي 2' : 'Please enter an integer >= 2'); return; } if (n > 10000000) { alert(currentLang === 'ar' ? '⚠️ عذراً فدوى! 50 مليون أو أكثر ثقيلة جداً على المتصفح وتتطلب محرك Python على الكمبيوتر. الحد الأقصى المتساهل هنا هو 10,000,000.' : '⚠️ 50M+ is too heavy for the browser and requires Python on a PC. Max limit here is 10,000,000.'); return; } const btn = document.getElementById('t-btn'); const progressContainer = document.getElementById('progress-container'); const progressBar = document.getElementById('progress-bar'); const progressText = document.getElementById('progress-text'); const resDiv = document.getElementById('results'); btn.disabled = true; progressContainer.style.display = 'block'; progressBar.style.width = '10%'; progressText.innerText = translations[currentLang].calculating; resDiv.innerText = "..."; const workerCode = ` function isPrime(num) { if (num < 2) return false; if (num === 2) return true; if (num % 2 === 0) return false; for (let i = 3; i <= Math.sqrt(num); i += 2) { if (num % i === 0) return false; } return true; } self.onmessage = function(e) { const n = e.data; let bridges = []; let step = Math.max(1, Math.floor(n / 10)); for (let d = 0; d < n; d++) { let p1 = n - d; let p2 = n + d; if (isPrime(p1) && isPrime(p2)) { bridges.push({ d, p1, p2 }); } if (d % step === 0) { let percent = Math.floor((d / n) * 90); self.postMessage({ type: 'progress', percent: percent }); } } self.postMessage({ type: 'done', bridges: bridges }); }; `; const blob = new Blob([workerCode], { type: 'application/javascript' }); const worker = new Worker(URL.createObjectURL(blob)); worker.onmessage = function(e) { if (e.data.type === 'progress') { progressBar.style.width = e.data.percent + '%'; } else if (e.data.type === 'done') { progressBar.style.width = '100%'; progressText.innerText = translations[currentLang].done; const bridges = e.data.bridges; let density = ((bridges.length / n) * 100).toFixed(2); let output = currentLang === 'ar' ? `H_${n}: ${bridges.length} جسر | الكثافة (Density): ${density}%\n\n` : `H_${n}: ${bridges.length} bridges | Density: ${density}%\n\n`; let displayBridges = bridges.length > 15 ? bridges.slice(0, 15) : bridges; displayBridges.forEach(b => { output += `d=${b.d} -> (${b.p1}, ${b.p2})\n`; }); if (bridges.length > 15) { output += `\n... (${currentLang === 'ar' ? 'تم إخفاء الباقي، الإجمالي' : 'hidden, total'}: ${bridges.length})`; } resDiv.innerText = output; btn.disabled = false; worker.terminate(); } }; worker.postMessage(n); }
